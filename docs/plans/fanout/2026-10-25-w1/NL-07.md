## 3. Shared drain helper (duplicated per app on purpose: apps may not import each other, and a shared package is A0/NL-01's)

`apps/<app>/src/drain.ts` (identical text in both, only the file header differs):

```ts
import type { Logger } from "@ls/obs/logger";

export const DRAIN_GRACE_MS = 25000; // [A] ticket assumption

export interface InflightTracker {
  /** Wrap a handler; resolves with the handler's result. Throws DrainingError once draining began. */
  track<T>(work: () => Promise<T>): Promise<T>;
  readonly draining: boolean;
  /** Stop accepting: after this, track() throws. Resolves when in-flight count is 0. */
  begin_drain(): Promise<void>;
  readonly in_flight: number;
}
export class DrainingError extends Error { constructor() { super("draining"); this.name = "DrainingError"; } }
export function create_inflight_tracker(): InflightTracker;

export interface SignalHost { on(sig: "SIGTERM" | "SIGINT", fn: () => void): unknown; off?(sig: "SIGTERM" | "SIGINT", fn: () => void): unknown }
export interface DrainOptions {
  tracker: InflightTracker;
  close: () => Promise<void>;          // closes server(s); called after in-flight settles or grace expires
  exit: (code: number) => void;        // injected; production passes process.exit
  logger: Logger;
  grace_ms?: number;                   // default DRAIN_GRACE_MS
  host?: SignalHost;                   // default process
  setTimer?: typeof setTimeout; clearTimer?: typeof clearTimeout; // fake-timer friendly (vi.useFakeTimers works without these)
}
/** Registers SIGTERM+SIGINT. First signal: log `process.drain` {extra:{signal}}, tracker.begin_drain(), race with a grace timer, await close(), exit(0).
 *  Second signal at any time: exit(1) at once. Returns an uninstall function. */
export function install_drain(opts: DrainOptions): () => void;
```
Behaviour: the first signal sets draining (no new work accepted), waits for `in_flight` to reach 0 or `grace_ms`, then `await close()`, then `exit(0)` (grace expiry still exits 0 after logging
`process.drain_timeout`). Idempotent: signals after the second are ignored. `close()` rejection logs `process.drain_failed` and exits 1.
